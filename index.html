إعدادات استلام وحفظ بيانات أولياء الأمور

كيف تعمل هذه الصفحة عند نشرها لأولياء الأمور؟

1. الحفظ التلقائي في Google Sheets (موصى به): عند وضع رابط Google Apps Script، يقوم زر "إرسال وحفظ البيانات" بترحيل جميع حقول الطالب مباشرة إلى جدول Excel/Sheets منظم لديك.

2. الإرسال الفوري عبر واتساب + تحميل ملف CSV/Excel: حتى بدون أي ربط خارجي، يستطيع ولي الأمر إرسال ملخص مرتب عبر واتساب بضغطة زر أو تنزيل ملف جدول بيانات (CSV) جاهز لإرساله لك.

كود Google Apps Script (انسخه والصقه في Extensions > Apps Script داخل جدول Google Sheets): 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  var headers = [
    "تاريخ الإرسال", "العام الدراسي", "الاسم", "اسم الأب", "اسم والد الأب", "اسم جد الأب", "اللقب",
    "تاريخ التولد", "اسم الأم", "اسم والد الأم", "اسم جد الأم", "نوع الهوية",
    "رقم الموحدة / الجنسية", "بلد الإصدار", "السجل", "الصحيفة", "الرقم الإحصائي",
    "الجنس", "الجنسية", "دولة التولد", "مسقط الرأس", "اللغة الأم", "فئة الدم",
    "الديانة", "رقم الهاتف", "لغة الدرس", "الاحتياجات الخاصة", "المستوى المعيشي",
    "منحة الرعاية", "الصف الدراسي", "الشعبة", "المدينة / القرية", "الحي",
    "المحلة", "الزقاق", "عنوان 1", "عنوان 2", "أقرب نقطة دالة", "البريد الإلكتروني",
    "اسم المنظم", "اسم ولي الأمر"
  ];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  }
  sheet.appendRow([
    new Date(), data.academic_year, data.first_name, data.father_name, data.grandfather_name,
    data.great_grandfather_name, data.surname, data.dob, data.mother_name,
    data.mother_father_name, data.mother_grandfather_name, data.id_type_label,
    data.id_number, data.issue_country, data.record, data.page, data.stat_number,
    data.gender_label, data.nationality, data.birth_country, data.birth_place,
    data.mother_tongue, data.blood_type, data.religion, data.phone, data.study_lang,
    data.special_needs_label, data.living_level_label, data.welfare_label, data.grade,
    data.section, data.city, data.neighborhood, data.mahalla, data.zuqaq,
    data.address1, data.address2, data.landmark, data.email, data.organizer, data.guardian
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
